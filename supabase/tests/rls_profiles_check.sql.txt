values
  ('00000000-0000-0000-0000-0000000000a1', '00000000-0000-0000-0000-000000000000', 'authenticated', 'authenticated', 'a@example.com',
   '{"provider":"email"}', '{"username":"Alice_1","marketing_opt_in":"true"}'),
  ('00000000-0000-0000-0000-0000000000b2', '00000000-0000-0000-0000-000000000000', 'authenticated', 'authenticated', 'b@example.com',
   '{"provider":"google"}', '{"username":"ignored"}');

select id, username, email, email_preferences from public.profiles order by email;
-- expect: a -> username 'alice_1', email_preferences true; b -> username NULL (OAuth never takes metadata)

set local role authenticated;
select set_config('request.jwt.claims', '{"sub":"00000000-0000-0000-0000-0000000000a1","role":"authenticated"}', true);

select count(*) as visible_profiles from public.profiles;                    -- expect 1
update public.profiles set username = 'hacked' where id = '00000000-0000-0000-0000-0000000000b2';  -- expect 0 rows
update public.profiles set email = 'x@example.com' where id = '00000000-0000-0000-0000-0000000000a1'; -- expect permission denied
